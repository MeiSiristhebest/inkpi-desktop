import { afterEach, describe, expect, it, vi } from 'vitest'
import { fireEvent, render, screen, waitFor } from '@testing-library/react'
import type { AiTask, TaskResult } from '@inkpi/protocol'
import type { ChapterRecord } from '../types'
import { DesktopPluginHostProvider, usePluginHostContext } from '../core/pluginHostContext'
import { indexedDbProjectRepository } from '../adapters/indexedDbProjectRepository'
import { indexedDbReaderHookRepository } from '../adapters/indexedDbReaderHookRepository'
import { ReaderHookMasterView } from '../plugins/reader-hook/components/ReaderHookMasterView'
import { MultiCalendarEngine } from '../plugins/multi-calendar/engine/MultiCalendarEngine'
import type { ChapterChronologyEvent } from '../plugins/multi-calendar/types'
import { projectChapterSequence } from '../domain/chapter/chapterSequence'
import { buildGroundedPluginInput } from '../domain/pluginAnalysis'
import { BrainstormSparkEngine } from '../plugins/brainstorm-spark/engine/BrainstormSparkEngine'

const projectId = 'plugin-review-audit-only'
function chapter(id: string, volumeId: string, order: number, content: string): ChapterRecord {
  return {
    id,
    projectId,
    volumeId,
    order,
    title: id,
    content,
    revision: 7,
    wordCount: content.length,
    updatedAt: 1,
  }
}

afterEach(() => vi.restoreAllMocks())

describe('Temporary read-only audit reproductions (assert current defects, not fixes)', () => {
  it('records the active chapter in task scope even when the plugin input selects a different chapter', async () => {
    const active = chapter('chapter-active-A', 'volume-a', 1, '甲章正文')
    const selected = chapter('chapter-selected-B', 'volume-b', 1, '乙章正文')
    let host: ReturnType<typeof usePluginHostContext> | undefined
    function Probe() {
      host = usePluginHostContext()
      return null
    }
    const submitted: AiTask[] = []
    const onAiTask = vi.fn(async (task: AiTask) => {
      submitted.push(task)
      return { taskId: task.id, status: 'completed', output: { format: 'text', text: 'audit result' } } as TaskResult
    })
    render(
      <DesktopPluginHostProvider projectId={projectId} activeChapter={active} chapters={[active, selected]} onAiTask={onAiTask} isAiConnected>
        <Probe />
      </DesktopPluginHostProvider>,
    )
    const input = buildGroundedPluginInput('reader-hook', {
      chapter: { id: selected.id, revision: selected.revision },
      endingText: selected.content,
    })!
    const outcome = await host!.aiAssistant!.runPluginOutcome!('reader-hook', input)
    expect(submitted[0].scope?.document?.id).toBe(active.id)
    expect(submitted[0].input.documentId).toBe(active.id)
    expect(input.input.chapter).toMatchObject({ id: selected.id })
    expect(outcome?.provenance.documentId).toBe(active.id)
    console.log('AUDIT_SCOPE_MISMATCH', JSON.stringify({ selected: selected.id, scope: submitted[0].scope?.document?.id, provenance: outcome?.provenance.documentId }))
  })

  it('can produce a false chronology regression from per-volume chapter order', () => {
    const chapters = [chapter('volume1-ch2', 'v1', 2, '第二十天'), chapter('volume2-ch1', 'v2', 1, '第三十天')]
    const volumes = [{ id: 'v1', projectId, title: '卷一', order: 1 }, { id: 'v2', projectId, title: '卷二', order: 2 }]
    const events = chapters.map((c, index) => ({
      id: `event-${c.id}`,
      chapterId: c.id,
      chapterTitle: c.title,
      chapterOrder: c.order,
      sequenceRole: 'mainline',
      timePoint: { absoluteDayIndex: index === 0 ? 20 : 30 },
    })) as unknown as ChapterChronologyEvent[]
    const actual = MultiCalendarEngine.validateChronology(events)
    const sequence = projectChapterSequence(projectId, chapters, volumes)
    const corrected = MultiCalendarEngine.validateChronology(events.map((event) => ({ ...event, chapterOrder: sequence.find((c) => c.id === event.chapterId)!.chapterNumber! })))
    expect(actual.hasConflict).toBe(true)
    expect(corrected.hasConflict).toBe(false)
    console.log('AUDIT_CROSS_VOLUME_FALSE_POSITIVE', JSON.stringify({ actual: actual.diagnostic, corrected: corrected.diagnostic }))
  })

  it('saves an edited hook copy with the original chapter revision but no edited-copy marker', async () => {
    const source = chapter('original-hook', 'v1', 1, '保存版本的原始正文。')
    vi.spyOn(indexedDbProjectRepository, 'getChaptersByProject').mockResolvedValue([source])
    vi.spyOn(indexedDbProjectRepository, 'getVolumesByProject').mockResolvedValue([{ id: 'v1', projectId, title: '卷一', order: 1 }])
    vi.spyOn(indexedDbReaderHookRepository, 'getAll').mockResolvedValue([])
    const save = vi.spyOn(indexedDbReaderHookRepository, 'save').mockResolvedValue()
    render(<ReaderHookMasterView projectId={projectId} />)
    const editor = screen.getByLabelText('待审章节正文')
    await waitFor(() => expect(editor).toHaveValue(source.content))
    fireEvent.change(editor, { target: { value: '这段内容从未出现在保存版本中。' } })
    fireEvent.click(screen.getByRole('button', { name: '入库本章断章' }))
    await waitFor(() => expect(save).toHaveBeenCalledTimes(1))
    const record = save.mock.calls[0][0]
    expect(record.sourceRevision).toBe(source.revision)
    expect(record.hookText).not.toBe(source.content)
    expect(record).not.toHaveProperty('sourceKind')
    console.log('AUDIT_EDITED_COPY_REVISION', JSON.stringify({ revision: record.sourceRevision, editedText: record.hookText, originalText: source.content }))
  })

  it('returns the same eight brainstorming operators for unrelated dilemmas rather than solving constraints', () => {
    const shared = { dilemmaType: 'combat' as const, currentSituation: '当前局势', protagonistGoal: '目标', enemyAdvantage: '对手条件' }
    const first = BrainstormSparkEngine.generateSolutions({ ...shared, coreProblem: '密室中找到失踪者', knownResources: ['钥匙'], constraints: ['不能离开密室'] })
    const second = BrainstormSparkEngine.generateSolutions({ ...shared, coreProblem: '避免两个商团价格战', knownResources: ['合同'], constraints: ['没有第三方'] })
    expect(first.map((item) => item.operatorId)).toEqual(second.map((item) => item.operatorId))
    expect(first).toHaveLength(8)
    expect(second).toHaveLength(8)
    console.log('AUDIT_BRAINSTORM_TEMPLATE_ONLY', JSON.stringify({ options: first.length, identicalOperators: true, sample: second[2].concretePlot }))
  })
})
